const digest=(bytes:Uint8Array)=>createHash("sha256").update(bytes).digest("hex");
function realDirectory(path:string){
  const root=resolve(path);if(!/^[A-Za-z]:[\\/]/.test(root))throw new Error("Use an absolute local browser directory.");
  let at=root.slice(0,3);for(const part of ["",...root.slice(3).split(sep)]){if(part)at=join(at,part);const stat=lstatSync(at);
    if(!stat.isDirectory()||stat.isSymbolicLink())throw new Error("Browser directories must be real and unlinked.");const canonical=realpathSync(at);
    // Bun's exact drive-root spelling difference is not a general path alias.
    const driveRoot=/^[A-Za-z]:\\$/.test(at)&&canonical===at.slice(0,-1);
    if(canonical!==at&&!driveRoot)throw new Error("Browser directories must be real and unlinked.");}
  return root;
}
/** Use a dedicated new process. Caller has restored into a new session directory
 * and supplies its captured test owner in memory. Never reads an existing private
 * identity/owner file. */
export async function launchPreservedOwner(restoredRoot:string,owner:CapturedOwner,capture:CapturedSeed,resume=false){

  if(dirname(session)!==parent||!/^s-[a-f0-9]{16}$/.test(session.slice(parent.length+1))||root!==join(session,"r"))throw new Error("Use the exact new short H-drive session restore directory.");
  if(owner.profile!=="dub-studio-fixture/1"||!owner.token||owner.token.length>4096||!/^[A-Za-z0-9_-]{1,128}$/.test(owner.projectId))throw new Error("Supply the captured test owner in memory.");

  owner={...owner};capture=structuredClone(capture);
  const worktree=resolve(import.meta.dir,"../../hollywood-video-current-mixed-source-v4");

  const frontend=join(worktree,"packages/frontend/src/index.html"),frontendSha256=digest(readFileSync(frontend)),sessionPath=join(session,"session.public.json"),stopPath=join(session,"stop"),restartPath=join(session,"restart");
  if(existsSync(stopPath))throw new Error("The stopped session needs an explicit new restore or reviewed stop-file reset.");
  // Exact public test-only profile from dub-studio.ts; never a production secret.
  const config={HV_TOKEN_SECRET:"dub-studio-fixture-secret-at-least-thirty-two-characters",HV_ANIMATIC_PROVIDER_POOL:'["mock"]',HV_PROVIDER_POOL:'["mock"]',HV_NARRATION:"1",HV_ANIMATIC_CAPTIONS:"0"};
  const previous=Object.fromEntries(Object.keys(config).map(key=>[key,process.env[key]]));Object.assign(process.env,config);
  let api:ApiServer|undefined,web:ReturnType<typeof Bun.serve>|undefined,stoppingApi:Promise<void>|undefined;

    if(resume){

      if(saved.schema!=="hv-private-preserved-owner-session/1"||saved.root!==root||saved.projectId!==owner.projectId||saved.worktree!==worktree||saved.frontendSha256!==frontendSha256||JSON.stringify(saved.capture)!==JSON.stringify(capture))throw new Error("Resume only this previously opened mutable browser session and selected capture.");
      // Browser saves change live files. The immutable seed's snapshot manifest
      // must not be revalidated against those legitimately changed files.
    }else{
      if(existsSync(sessionPath))throw new Error("The browser session already has a launch record.");
      const snapshot=readStateSnapshot(root);
      if(digest(Buffer.from(JSON.stringify(snapshot)))!==capture.stateSha256)throw new Error("The restored state differs from the exact selected A/B capture.");

  }catch(error){try{await stopServers();}finally{restoreEnvironment();}throw error;}
}
